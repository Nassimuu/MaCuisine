      { id: 3, name: "Robot Pâtissier Multifonction 1200W", price: 14200, rating: 4.7, reviews: 98, img: "https://images.unsplash.com/photo-1578643463396-0997cb5328c1?w=500&q=80" },
      { id: 4, name: "Set de 5 Couteaux de Chef Professionnels en Acier", price: 3400, rating: 4.6, reviews: 142, img: "https://images.unsplash.com/photo-1593618998160-e34014e67546?w=500&q=80" },
      { id: 5, name: "Mixeur Plongeant 4 en 1 avec Hachoir & Fouet", price: 4200, rating: 4.5, reviews: 76, img: "https://images.unsplash.com/photo-1590794056226-79ef3a8147e1?w=500&q=80" },
      { id: 6, name: "Balance de Cuisine Électronique Précision 1g/5kg", price: 1500, rating: 4.8, reviews: 189, img: "https://images.unsplash.com/photo-1585515320310-259814833e62?w=500&q=80" }
    ];

    let cart = [];

    function renderProducts(items = products) {
      const container = document.getElementById('productGrid');
      if (items.length === 0) {
        container.innerHTML = `<p class="col-span-4 text-center text-gray-500 py-8">Aucun produit trouvé.</p>`;
        return;
      }
      container.innerHTML = items.map(p => `
        <div class="bg-white p-4 rounded-lg shadow-sm hover:shadow-md transition flex flex-col justify-between border border-gray-200">
          <div>
            <div class="h-44 w-full mb-3 flex items-center justify-center overflow-hidden bg-gray-50 rounded">
              <img src="${p.img}" alt="${p.name}" class="h-full object-cover">
            </div>
            <h3 class="font-semibold text-sm line-clamp-2 hover:text-[#007185] cursor-pointer">${p.name}</h3>
            <div class="flex items-center gap-1 text-xs text-[#de7921] my-1">
              <span>★ ${p.rating}</span>
              <span class="text-gray-500">(${p.reviews} avis)</span>
            </div>
            <div class="text-xl font-black text-gray-900 mt-2">
              ${p.price.toLocaleString()} <span class="text-xs font-normal">DA</span>
            </div>
          </div>
          <button onclick="addToCart(${p.id})" class="mt-4 w-full bg-[#ffd814] hover:bg-[#f7ca00] text-gray-900 text-xs font-bold py-2.5 px-3 rounded-full shadow-sm active:scale-95 transition">
            Ajouter au panier
          </button>
        </div>
      `).join('');
    }

    function addToCart(id) {
      const item = products.find(p => p.id === id);
      cart.push(item);
      updateCart();
    }

    function updateCart() {
      document.getElementById('cartCount').innerText = cart.length;
      const cartItems = document.getElementById('cartItems');
      const cartTotal = document.getElementById('cartTotal');
      
      if (cart.length === 0) {
        cartItems.innerHTML = '<p class="text-sm text-gray-500">Votre panier est vide.</p>';
        cartTotal.innerText = '0 DA';
        return;
      }

      let total = 0;
      cartItems.innerHTML = cart.map((p, index) => {
        total += p.price;
        return `
          <div class="flex justify-between items-center text-sm border-b pb-2">
            <div>
              <p class="font-medium text-gray-800 line-clamp-1">${p.name}</p>
              <p class="text-xs text-gray-500">${p.price.toLocaleString()} DA</p>
            </div>
            <button onclick="removeFromCart(${index})" class="text-red-500 text-xs font-semibold hover:underline">Supprimer</button>
          </div>
        `;
      }).join('');
      
      cartTotal.innerText = `${total.toLocaleString()} DA`;
    }

    function removeFromCart(index) {
      cart.splice(index, 1);
      updateCart();
    }

    function openCart() {
      document.getElementById('cartModal').classList.remove('hidden');
    }

    function closeCart() {
      document.getElementById('cartModal').classList.add('hidden');
    }

    function filterProducts() {
      const query = document.getElementById('searchInput').value.toLowerCase();
      const filtered = products.filter(p => p.name.toLowerCase().includes(query));
      renderProducts(filtered);
    }

    function checkout() {
      if (cart.length === 0) {
        alert("Votre panier est vide !");
        return;
      }
      alert("Merci pour votre commande ! Nous vous contacterons pour confirmer la livraison.");
    }

    renderProducts();
  </script>
